// Serialize integrations across lanes.
let lock = Promise.resolve()
function serial(fn) { const p = lock.then(fn); lock = p.then(() => {}, () => {}); return p }

const results = {}
const failed = []

async function runGame(year, name, port, lane, retryNote) {
  const game = `${year}/${name}`
  const dir = `apps/games/${year}/${name}`
  const tag = retryNote ? 'retry' : 'Games'
  const impl = await agent(implPrompt(game, dir, port, retryNote), { label: `impl:${name}`, phase: tag, schema: IMPL_SCHEMA })
  if (!impl) { return { game, stage: 'impl', ok: false, note: 'implementer returned nothing' } }
  const rev = await agent(reviewPrompt(game, dir, port, impl), { label: `review:${name}`, phase: tag, schema: REVIEW_SCHEMA })
  if (!rev || rev.verdict !== 'committed' || !rev.commit) {
    return { game, stage: 'review', ok: false, impl, rev, note: JSON.stringify(rev ? rev.remaining : 'no review result') }
  }
  const integ = await serial(() => agent(integratePrompt(game, rev), { label: `integrate:${name}`, phase: tag, schema: INTEGRATE_SCHEMA, effort: 'medium' }))
  if (!integ || integ.status !== 'integrated') {
    return { game, stage: 'integrate', ok: false, impl, rev, integ, note: JSON.stringify(integ || 'no integration result') }
  }
  return { game, ok: true, impl, rev, integ }
}
